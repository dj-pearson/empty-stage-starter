# ============================================================================
# App Store Listings (optional, but recommended for SEO)
# ============================================================================
# Numeric Apple App Store ID, taken from the listing URL:
#   https://apps.apple.com/us/app/eatpal/id<APP_ID>
# When set, the site links to the App Store listing and adds installUrl/sameAs to the
# structured data, which is how search engines treat the website and the app listing as
# one entity. Unset = every store link and store schema field is simply omitted.
VITE_APP_STORE_APP_ID=
# It also drives the iOS Smart App Banner (the apple-itunes-app meta tag in
# src/components/SEOHead.tsx). Mobile is where EatPal's search traffic converts,
# so leaving this unset costs the best-performing device its route into the app.
# Android package name, e.g. com.eatpal.app
VITE_PLAY_STORE_PACKAGE=

# 10-character Apple Team ID from developer.apple.com, e.g. A1B2C3D4E5. Build-time only,
# so it is NOT prefixed with VITE_ and never reaches the client bundle.
# scripts/generate-aasa.mjs uses it to write
# dist/.well-known/apple-app-site-association, which is what makes Universal Links work.
# The Swift app has claimed applinks:tryeatpal.com in its entitlements since it shipped,
# but the website never served the matching file, so tapping a tryeatpal.com link on an
# iPhone opens Safari and never the app. Unset = the file is not written and Universal
# Links stay off; a wrong value is worse than none, because Apple caches what it fetches.
APPLE_TEAM_ID=
# Overrides the bundle ID in the association file. Defaults to com.eatpal.app, which
# matches PRODUCT_BUNDLE_IDENTIFIER in ios/EatPal/project.yml.
APPLE_BUNDLE_ID=
